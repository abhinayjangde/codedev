Time: O(n)
Space: O(1)

solution in python
```python
class Solution:
    def maxSubarraySum(self, arr, k):
        # code here
        max_sum = 0
        for i in range(0,k):
            max_sum += arr[i]
            
        current_sum = max_sum
        
        for i in range(0, len(arr) - k):
            current_sum -= arr[i]
            current_sum += arr[i+k]
            
            max_sum = max(max_sum, current_sum)
        
        return max_sum
```

solution in c++
```cpp
class Solution {
  public:
    int maxSubarraySum(vector<int>& arr, int k) {
        // code here
        int max_sum = 0;
        
        for(int i=0; i<k; i++){
            max_sum += arr[i];
        }
        
        // slide window
        int curr = max_sum;
        
        for(int i=0; i<arr.size()-k; i++){
            curr -= arr[i];
            curr += arr[i+k];
            
            if(curr > max_sum){
                max_sum = curr;
            }
        }
        
        return max_sum;
    }
};
```
